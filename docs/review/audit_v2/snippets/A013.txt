    def visual_inspection_scan():
        # Move to home position
        movej(p[0, -90, 120, 0, -90, 0], a=1.4, v=1.05)
    
        # Approach inspection start position (above surface)
        movel(p[0, -90, 180, 0, -90, 0], a=1.2, v=1.02)
    
        # Descend to scan height (150 mm above part)
        movel(p[0, -90, 160, 0, -90, 0], a=1.2, v=0.8)
    
        # Scan line 1 — Y sweep
        movel(p[-100, -90, 160, 0, -90, 0], a=1.2, v=0.5)
        
        # Step to scan line 2 (+X)
        movel(p[0, -80, 160, 0, -90, 0], a=1.2, v=0.8)
    
        # Scan line 2 — Y sweep
        movel(p[-100, -80, 160, 0, -90, 0], a=1.2, v=0.5)
        
        # Step to scan line 3 (+X)
        movel(p[0, -70, 160, 0, -90, 0], a=1.2, v=0.8)
    
        # Scan line 3 — Y sweep
        movel(p[-100, -70, 160, 0, -90, 0], a=1.2, v=0.5)
        
        # Step back to scan line 2 (-X)
        movel(p[0, -80, 160, 0, -90, 0], a=1.2, v=0.8)
    
        # Step back to scan line 1 (-X)
        movel(p[0, -90, 160, 0, -90, 0], a=1.2, v=0.8)
    
        # Ascend from scan height
        movel(p[0, -90, 180, 0, -90, 0], a=1.2, v=0.8)
    
        # Move back to home position
        movej(p[0, -90, 120, 0, -90, 0], a=1.4, v=1.05)
    end
    
    visual_inspection_scan()
