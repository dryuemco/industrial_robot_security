    # Define home position
    home_pos = [0, -1.57, 1.57, -1.83, -1.57, 0]
    
    # Define conveyor pick positions
    conveyor_pick_above = p[0.6, -0.2, 0.4, 3.14, 0, 0]
    conveyor_pick_descend = p[0.6, -0.2, 0.1, 3.14, 0, 0]
    
    # Define pallet positions
    pallet_pos_00 = [0.5, 0.5, 0.1, 3.14, 0, 0]
    pallet_pos_01 = [0.7, 0.5, 0.1, 3.14, 0, 0]
    pallet_pos_10 = [0.5, 0.7, 0.1, 3.14, 0, 0]
    pallet_pos_11 = [0.7, 0.7, 0.1, 3.14, 0, 0]
    
    # Move to home position
    movej(home_pos, a=1.2, v=0.5)
    
    # Loop for picking and placing 4 boxes
    for i from 0 to 3 do
        # Approach conveyor pick above
        movel(conveyor_pick_above, a=0.8, v=0.25)
        
        # Descend to pick position
        movel(conveyor_pick_descend, a=0.1, v=0.1)
        
        # Simulate picking (open and close gripper)
        popup("Picking box", 3)
        sleep(1)  # Simulated time for gripping
        
        # Lift from conveyor
        movel(conveyor_pick_above, a=0.8, v=0.25)
        
        # Transit to pallet position with fenced mode
        if i == 0 then
            movel(pallet_pos_00, a=1.4, v=0.28)
        elif i == 1 then
            movel(pallet_pos_01, a=1.4, v=0.28)
        elif i == 2 then
            movel(pallet_pos_10, a=1.4, v=0.28)
        else
            movel(pallet_pos_11, a=1.4, v=0.28)
        endif
        
        # Simulate placing (open and close gripper)
        popup("Placing box", 3)
        sleep(1)  # Simulated time for releasing
        
        # Return to conveyor pick above
        movel(conveyor_pick_above, a=0.8, v=0.25)
    endfor
    
    # Move back to home position
    movej(home_pos, a=1.2, v=0.5)
    
    popup("Task completed", 3)
